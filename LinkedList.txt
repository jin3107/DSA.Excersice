public class Node
{
    public object Value;
    public Node Next;

    public Node()
    { 
        Value = null;
        Next = null;
    }

    public Node(object val)
    {
        Value = val;
        Next = null;
    }
}

    public void PrintList()
    {
        if (First == null)
        {
            Console.WriteLine("Danh sách rỗng");
            return;
        }
        Node curNode = First;
        while (curNode != null)
        {
            Console.Write(curNode.Value);
            if (curNode.Next != null)
                Console.Write(" -> ");
            curNode = curNode.Next;
        }
    }

    // 2. Thêm X vào sau V
    public Node AddXAfterV(object X, object V)
    {
        if (First == null || X == null || V == null)
            return null;
        Node curNode = First;
        while (curNode != null && !curNode.Value.Equals(V))
            curNode = curNode.Next;
        if (!curNode.Value.Equals(V))
            return null;
        Node newNode = new Node(X);
        newNode.Next = curNode.Next;
        curNode.Next = newNode;
        return newNode;
    }

    // 3. Thêm X vào trước V
    public Node AddXBeforeV(object X, object V)
    {
        if (First == null || X == null || V == null) return null;
        Node curNode = First;
        Node prevNode = null;
        while (curNode != null && !curNode.Value.Equals(V))
        {
            prevNode = curNode;
            curNode = curNode.Next;
        }
        if (curNode == null) return null;
        Node newNode = new Node(X);
        if (prevNode == null)
        {
            newNode.Next = First;
            First = newNode;
        }
        else
        {
            prevNode.Next = newNode;
            newNode.Next = curNode;
        }
        return newNode;
    }

    // 22. Xóa giá trị X trong dslk
    public void Remove(object X)
    {
        if ((First == null || (X == null))) return;
        if (First.Value.Equals(X))
        {
            First = First.Next;
            return;
        }

        Node curNode = First;
        while ((curNode.Next != null) && !curNode.Next.Value.Equals(X))
            curNode = curNode.Next;
        if (curNode.Next != null)
            curNode.Next = curNode.Next.Next;
        Console.WriteLine($"Đã xóa thành công {X}");
    }

    // 24. Xóa phần tử kề sau phần tử Q trong dslk
    public void RemoveAfter(object Q)
    {
        if (First == null || Q == null) return;
        Node curNode = First;
        if (First == Q)
        {
            First = First.Next;
            return;
        }
        while (curNode != null && (int)curNode.Value != (int)Q)
            curNode = curNode.Next;
        if (curNode != null && curNode.Next != null)
            curNode.Next = curNode.Next.Next;
    }

    // 25. Xóa phần tử kề trước phần tử Q trong dslk
    public void RemoveBefore(object Q)
    {
        if (First == null || Q == null || (int)First.Value == (int)Q) return;
        Node cur = First;
        Node prev = null;
        Node prePrev = null;
        while (cur != null && (int)cur.Value != (int)Q)
        {
            prePrev = prev;
            prev = cur;
            cur = cur.Next;
        }
        if (prev == null) return;
        if (prePrev == null)
            First = cur;
        else
            prePrev.Next = cur;
    }

    public void RemoveCenter()
    {

        Node prevNode = null;
        int count = 0;
        while (curNode != null)
        {
            count++;
            curNode = curNode.Next;
        }
        curNode = First;
        int mid = count / 2;
        for (int i = 0; i < mid; i++)
        {
            prevNode = curNode;
            curNode = curNode.Next;
        }
        if (prevNode != null)
            prevNode.Next = curNode.Next;
    }

    private bool IsPrime(int number)
    {
        if (number < 2) return false;
        for (int i = 2; i <= Math.Sqrt(number); i++)
        {
            if (number % i == 0) return false;
        }
        return true;
    }
