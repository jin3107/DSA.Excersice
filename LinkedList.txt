public void PrintList()
{

    {
        Console.WriteLine("Danh sách rỗng.");
        return;
    }

    Node<T> curNode = First;
    while (curNode != null)
    {
        Console.Write(curNode.Value);
        if (curNode.Next != null)
            Console.Write(" -> ");
        curNode = curNode.Next;
    }
    Console.WriteLine();
}
