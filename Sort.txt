public void Swap<T>(ref T a, ref T b)
{
    T temp = a;
    a = b;
    b = temp;
}

public void InsertionSort(int[] arr, int n)
{
    //int k = 1;
    for (int i = 1; i < n; i++)
    {
        int key = arr[i];
        int j = i - 1;

        while (j >= 0 && arr[j] > key)
        {
            arr[j + 1] = arr[j];
            j--;
        }
        arr[j + 1] = key;
        //Console.WriteLine($"Vòng lặp {k++}: " + string.Join(", ", arr));
    }
}

// Sắp xếp nhanh bằng PP đệ quy (Tăng dần)
public void QuickSort(int[] arr, int left, int right)
{
    if (left < right)
    {
        int pivot = arr[(left + right) / 2];
        int i = left, j = right;
        while (i <= j)
        {
            while (arr[i] < pivot) i++;
            while (arr[j] > pivot) j--;
            if (i <= j)
            {
                Swap(ref arr[i], ref arr[j]);
                i++;
                j--;
            }
        }
        QuickSort(arr, left, j);
        QuickSort(arr, i, right);
    }
}

// Sắp xếp nhanh bằng PP lặp (Tăng dần)
public void QuickSort(int[] arr, int left, int right)
{
    Stack<int> stack = new Stack<int>();
    stack.Push(left);
    stack.Push(right);
    while (stack.Count > 0)
    {
        right = stack.Pop();
        left = stack.Pop();
        int pivot = arr[(left + right) / 2];
        int i = left, j = right;
        while (i <= j)
        {
            while (arr[i] < pivot) i++;
            while (arr[j] > pivot) j--;
            if (i <= j)
            {
                Swap(ref arr[i], ref arr[j]);
                i++;
                j--;
            }
        }
        if (left < j) { stack.Push(left); stack.Push(j); }
        if (i < right) { stack.Push(i); stack.Push(right); }
    }
}

// Sắp xếp vun đống (Tăng dần)
public void AdjustHeap(int[] arr, int iRoot, int iEnd)
{
    int saved = arr[iRoot];
    int i = iRoot;
    int j = 2 * i + 1;
    while (j <= iEnd)
    {
        if ((j + 1 <= iEnd) && (arr[j] < arr[j + 1]))
            j += 1;
        if (arr[j] <= saved)
            break;
        else
        {
            arr[i] = arr[j];
            arr[j] = saved;
            i = j;
            j = 2 * i + 1;
        }
    }
    //Console.Write("\n\tHieu chinh dong day tu chi so {0} den {1}: ", iRoot, iEnd);
    //PrintArray(arr);
    //Console.WriteLine();
}

public void CreateHeap(int[] arr, int n)
{
    //Console.WriteLine("GIAI DOAN 1.TAO DONG CHO MANG\n");
    for (int iRoot = (n / 2) - 1; iRoot >= 0; iRoot--)
        AdjustHeap(arr, iRoot, n - 1);
}

public void HeapSort(int[] arr, int n)
{
    CreateHeap(arr, n);
    //Console.WriteLine("\nGIAI DOAN 2.LAP: HOAN VI 2 PHAN TU VA HIEU CHINH DONG\n");
    int iEnd = n - 1;
    while (iEnd > 0)
    {
        Swap(ref arr[0], ref arr[iEnd]);
        //Console.Write("\n\tHoan vi phan tu chi so 0 voi chi so {0}: ", iEnd);
        //PrintArray(arr);
        //Console.WriteLine();
        iEnd--;
        AdjustHeap(arr, 0, iEnd);
    }
}

// Sắp xếp trộn trực tiếp (Tăng dần)
public void Merge(int[] arrIN, int[] arrOUT, int start, int end1, int end2)
{
    int i = start;
    int j = end1 + 1;
    int o = start;
    while (i <= end1 && j <= end2)
    {
        if (arrIN[i] <= arrIN[j])
            arrOUT[o++] = arrIN[i++];
        else
            arrOUT[o++] = arrIN[j++];
    }
    while (i <= end1)
        arrOUT[o++] = arrIN[i++];
    while (j <= end2)
        arrOUT[o++] = arrIN[j++];
}

public void MergeSort(int[] arr, int n)
{
    //Console.Write($"MANG BAN DAU (run chieu dai = 1): \t");
    //PrintArray(arr);
    //Console.WriteLine();
    bool flag = true;
    int len = 1;
    int[] arrTemp = new int[n];
    while (len < n)
    {
        if (flag)
            MergeByLength(arr, arrTemp, n, len);
        else
            MergeByLength(arrTemp, arr, n, len);
        len *= 2;
        flag = !flag;
    }
    if (!flag)
        arr = arrTemp;
}
