public class Student
{
    public int Id { get; set; }
    public double M1 { get; set; }
    public double M2 { get; set; }
    public double M3 { get; set; }
    public double GetAverage() => (M1 + M2 + M3) / 3;
}

public class ProcessSort
{
    private List<Student> students;

    public ProcessSort(List<Student> students)
    {
        this.students = students;
    }

    public void InterchangeSortAscending()
    {
        int n = students.Count;
        for (int i = 0; i < n - 1; i++)
        {
            for (int j = i + 1; j < n; j++)
            {
                if (students[i].GetAverage() > students[j].GetAverage())
                {
                    (students[i], students[j]) = (students[j], students[i]);
                }
            }
        }
    }

    public void InterchangeSortDescending()
    {
        int n = students.Count;
        for (int i = 0; i < n - 1; i++)
        {
            for (int j = i + 1; j < n; j++)
            {
                if (students[i].GetAverage() < students[j].GetAverage())
                {
                    (students[i], students[j]) = (students[j], students[i]);
                }
            }
        }
    }

    public void BullbleSortAscending()
    {
        int n = students.Count;
        for (int i = n - 1; i > 0; i--)
        {
            for (int j = 0; j < i; j++)
            {
                if (students[j].GetAverage() < students[j - 1].GetAverage())
                {
                    (students[j], students[j - 1]) = (students[j - 1], students[j]);
                }
            }
        }
    }

    public void BullbleSortDescending()
    {
        int n = students.Count;
        for (int i = n - 1; i > 0; i--)
        {
            for (int j = 0; j < i; j++)
            {
                if (students[j].GetAverage() > students[j - 1].GetAverage())
                {
                    (students[j], students[j - 1]) = (students[j - 1], students[j]);
                }
            }
        }
    }

    public void ShakerSortAscending()
    {
        int n = students.Count;
        int left = 0;
        int right = n - 1;
        while (left <= right)
        {
            for (int i = left; i < right; i++)
            {
                if (students[i].GetAverage() > students[i + 1].GetAverage())
                {
                    (students[i], students[i + 1]) = (students[i + 1], students[i]);
                }
            }
            right--;
            for (int i = right; i > left; i--)
            {
                if (students[i].GetAverage() < students[i - 1].GetAverage())
                {
                    (students[i], students[i - 1]) = (students[i - 1], students[i]);
                }
            }
            left++;
        }
    }

    public void ShakerSortDescending()
    {
        int n = students.Count;
        int left = 0;
        int right = n - 1;
        while (left <= right)
        {
            for (int i = left; i < right; i++)
            {
                if (students[i].GetAverage() < students[i + 1].GetAverage())
                {
                    (students[i], students[i + 1]) = (students[i + 1], students[i]);
                }
            }
            right--;
            for (int i = right; i > left; i--)
            {
                if (students[i].GetAverage() > students[i - 1].GetAverage())
                {
                    (students[i], students[i - 1]) = (students[i - 1], students[i]);
                }
            }
            left++;
        }
    }

    public void SelectionSortAscending()
    {
        int n = students.Count;
        for (int i = 0; i < n - 1; i++)
        {
            int minIndex = i;
            for (int j = i + 1; j < n; j++)
            {
                if (students[j].GetAverage() < students[minIndex].GetAverage())
                    minIndex = j;
            }
            if (minIndex != i)
            {
                (students[i], students[minIndex]) = (students[minIndex], students[i]);
            }
        }
    }

    public void SelectionSortDescending()
    {
        int n = students.Count;
        for (int i = 0; i < n - 1; i++)
        {
            int maxIndex = i;
            for (int j = i + 1; j < n; j++)
            {
                if (students[j].GetAverage() > students[maxIndex].GetAverage())
                    maxIndex = j;
            }
            if (maxIndex != i)
            {
                (students[i], students[maxIndex]) = (students[maxIndex], students[i]);
            }
        }
    }

    public void InsertionSortAscending()
    {
        int n = students.Count;
        for (int i = 1; i < n; i++)
        {
            Student key = students[i];
            int j = i - 1;
            while (j >= 0 && students[j].GetAverage() > key.GetAverage())
            {
                students[j + 1] = students[j];
                j--;
            }
            students[j + 1] = key;
        }
    }

    public void InsertionSortDescending()
    {
        int n = students.Count;
        for (int i = 1; i < n; i++)
        {
            Student key = students[i];
            int j = i - 1;
            while (j >= 0 && students[j].GetAverage() < key.GetAverage())
            {
                students[j + 1] = students[j];
                j--;
            }
            students[j + 1] = key;
        }
    }

    public void QuickSortAscending(int low, int high)
    {
        if (low < high)
        {
            int pi = PartitionAscending(low, high);
            QuickSortAscending(low, pi - 1);
            QuickSortAscending(pi + 1, high);
        }
    }

    private int PartitionAscending(int low, int high)
    {
        double pivot = students[high].GetAverage();
        int i = low - 1;
        for (int j = low; j < high; j++)
        {
            if (students[j].GetAverage() < pivot)
            {
                i++;
                (students[i], students[j]) = (students[j], students[i]);
            }
        }
        (students[i + 1], students[high]) = (students[high], students[i + 1]);
        return i + 1;
    }

    public void QuickSortDescending(int low, int high)
    {
        if (low < high)
        {
            int pi = PartitionDescending(low, high);
            QuickSortDescending(low, pi - 1);
            QuickSortDescending(pi + 1, high);
        }
    }

    private int PartitionDescending(int low, int high)
    {
        double pivot = students[high].GetAverage();
        int i = low - 1;
        for (int j = low; j < high; j++)
        {
            if (students[j].GetAverage() > pivot)
            {
                i++;
                (students[i], students[j]) = (students[j], students[i]);
            }
        }
        (students[i + 1], students[high]) = (students[high], students[i + 1]);
        return i + 1;
    }
}

public class Student
{
    public string MaSinhVien { get; set; }
    public string HoTen { get; set; }
    public double DiemTrungBinh { get; set; }
    public string NganhHoc { get; set; }

    public Student(string maSinhVien, string hoTen, double diemTrungBinh, string nganhHoc)
    {
        MaSinhVien = maSinhVien;
        HoTen = hoTen;
        DiemTrungBinh = diemTrungBinh;
        NganhHoc = nganhHoc;
    }
}

public class Sort_Search_SinhVien
{
    public List<Student> students = new List<Student>();

    public void SelectionSortHoTen()
    {
        int n = students.Count;
        for (int i = 0; i < n - 1; i++)
        {
            int minIndex = i;
            for (int j = i + 1; j < n; j++)
            {
                if (string.Compare(students[j].HoTen, students[minIndex].HoTen) < 0)
                    minIndex = j;
            }
            if (minIndex != i)
            {
                (students[i], students[minIndex]) = (students[minIndex], students[i]);
            }
        }
    }

    public void BubbleSortMaSV()
    {
        int n = students.Count;
        for (int i = 0; i < n - 1; i++)
        {
            for (int j = 0; j < n - i - 1; j++)
            {
                if (string.Compare(students[j].MaSinhVien, students[j + 1].MaSinhVien) > 0)
                {
                    (students[j], students[j + 1]) = (students[j + 1], students[j]);
                }
            }
        }
    }

    public void QuickSortRecursiveDTB(int left, int right)
    {
        if (left < right)
        {
            double pivot = students[left].DiemTrungBinh;
            int i = left, j = right;
            while (i <= j)
            {
                while (students[i].DiemTrungBinh < pivot) i++;
                while (students[j].DiemTrungBinh > pivot) j--;
                if (i <= j)
                {
                    (students[i], students[j]) = (students[j], students[i]);
                    i++;
                    j--;
                }
            }
            QuickSortRecursiveDTB(left, j);
            QuickSortRecursiveDTB(i, right);
        }
    }

    public void BinSearchMaxDTB()
    {
        double max = students.Max(s => s.DiemTrungBinh);
        int left = 0, right = students.Count - 1;
        while (left <= right)
        {
            int mid = (left + right) / 2;
            if (students[mid].DiemTrungBinh == max)
            {
                Console.WriteLine("Sinh viên có điểm trung bình cao nhất:");
                students[mid].Xuat();
                return;
            }
            else if (students[mid].DiemTrungBinh < max)
                left = mid + 1;
            else
                right = mid - 1;
        }
    }

    public void BinSearchMinDTB()
    {
        double min = students.Min(s => s.DiemTrungBinh);
        int left = 0, right = students.Count - 1;
        while (left <= right)
        {
            int mid = (left + right) / 2;
            if (students[mid].DiemTrungBinh == min)
            {
                Console.WriteLine("Sinh viên có điểm trung bình thấp nhất:");
                students[mid].Xuat();
                return;
            }
            else if (students[mid].DiemTrungBinh < min)
                left = mid + 1;
            else
                right = mid - 1;
        }
    }

    public void SearchSurnamePopularity(List<Student> students)
    {
        if (students == null || students.Count == 0)
        {
            Console.WriteLine("Danh sách sinh viên rỗng");
            return;
        }

        Dictionary<string, int> surnameCount = new Dictionary<string, int>();
        foreach (var student in students)
        {
            string surname = GetSurname(student.HoTen);
            if (surnameCount.ContainsKey(surname))
                surnameCount[surname]++;
            else
                surnameCount[surname] = 1;
        }

        int maxCount = surnameCount.Values.Max();
        List<string> mostSurnames = surnameCount.Where(h => h.Value == maxCount).Select(h => h.Key).ToList();

        Console.WriteLine("\n===== HỌ PHỔ BIẾN NHẤT =====");
        foreach (var surname in mostSurnames)
        {
            Console.WriteLine($"Họ '{surname}' xuất hiện {maxCount} lần.");

            var studentsWithSurname = students.Where(s => GetSurname(s.HoTen) == surname);
            foreach (var s in studentsWithSurname)
            {
                Console.WriteLine($"- {s.HoTen} (Ngành: {s.NganhHoc})");
            }
        }
    }

    private string GetSurname(string fullname)
    {
        return fullname.Split(' ')[0];
    }
}
